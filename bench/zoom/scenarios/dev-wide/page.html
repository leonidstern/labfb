wait --pages: reading the list

The pages a wait listens on, as the lab writes them. It is in src/pages.mjs; node test.mjs checks it.

export function listenPages(raw) {
  if (!raw) return null;
  return String(raw).split(',').map((p) => '/' + p.trim().replace(/^\/+/, '').replace(/\.html?$/i, '')).filter((p) => p !== '/');
}
